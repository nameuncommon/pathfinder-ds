export function MotionDemo() {
  const [checked, setChecked] = useState(false)
  const fast = tokens.primitives.motion['duration/fast']
  const ease = tokens.primitives.motion['ease/standard']
  return (
    <div style={{ display: 'grid', gap: 'var(--space-3)' }}>
      <Checkbox checked={checked} onChange={setChecked} label="plan the next waypoint" />
      <p style={{ ...labelStyle, margin: 0 }}>
        {`Ticking it pops the check in over duration/fast (${fast}ms) with ease/standard (${ease}). With the Motion switch in the toolbar set to reduced, it appears instantly.`}
      </p>
    </div>
  )
}
