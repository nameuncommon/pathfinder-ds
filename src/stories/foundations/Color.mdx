export const primitiveEntries = Object.entries(tokens.primitives.color)

export const ramps = primitiveEntries.reduce((acc, [name, hex]) => {
  const ramp = name.split('/')[0]
  if (!acc[ramp]) acc[ramp] = []
  acc[ramp].push({ name, hex })
  return acc
}, {})

export function resolveColor(value) {
  const match = /^\{(.+)\}$/.exec(value)
  if (!match) return { token: null, hex: value }
  return { token: match[1], hex: tokens.primitives.color[match[1]] }
}
