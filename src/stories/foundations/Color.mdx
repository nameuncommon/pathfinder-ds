import { Meta, Unstyled } from '@storybook/addon-docs/blocks'
import tokens from '../../../tokens/tokens.json'
import { ThemeSync } from './ThemeSync'
import { cell, h1Style, h2Style, headCell, page, pStyle } from './docStyles'

<Meta title="Foundations/Colour" />

<ThemeSync />

export const prim = tokens.primitives.color
export const dark = tokens.semantic.dark
export const resolve = (v) => (v.startsWith('{') ? prim[v.slice(1, -1)] : v)

export const brand = [
  ['navy/950', 'The field. Every screen sits on it.'],
  ['navy/900', 'Brand navy: ink on paper, the logo on cream.'],
  ['navy/800', 'Raised surfaces and the solid glass fallback.'],
  ['gold/500', 'The one accent: the primary action, the logo, progress.'],
  ['gold/400', 'Hover and the focus ring.'],
  ['cream/100', 'Paper and primary text.'],
]
export const bearings = ['health', 'people', 'home', 'work', 'craft']
export const dormant = Object.keys(dark).filter((k) => k.startsWith('tag-text/'))
export const semantic = Object.keys(dark).filter((k) => !dormant.includes(k) && !k.startsWith('bearing/'))

export function Swatch({ hex, size = 40 }) {
  return (
    <span style={{ position: 'relative', display: 'inline-block', width: size, height: size, borderRadius: 8, background: 'var(--navy-950)', border: '1px solid var(--border-subtle)', overflow: 'hidden', flexShrink: 0 }}>
      <span style={{ position: 'absolute', inset: 0, background: hex }} />
    </span>
  )
}

<Unstyled>
  <div style={page}>
    <h1 style={h1Style}>Colour</h1>
    <p style={pStyle}>A calm navy field, gold as the single accent, cream for paper and text. Everything on this page is read from tokens.json, so it cannot drift from the system.</p>

    <h2 style={h2Style}>Brand</h2>
    <div style={{ display: 'grid', gridTemplateColumns: 'repeat(auto-fill, minmax(260px, 1fr))', gap: 'var(--space-5)', maxWidth: 900 }}>
      {brand.map(([k, role]) => (
        <div key={k} style={{ display: 'flex', gap: 'var(--space-3)', alignItems: 'flex-start' }}>
          <Swatch hex={prim[k]} size={56} />
          <div>
            <p style={{ margin: 0, fontWeight: 600, color: 'var(--text-primary)' }}>{k}</p>
            <p style={{ margin: '2px 0', fontFamily: 'var(--font-mono)', fontSize: 'var(--type-13-size)', color: 'var(--text-secondary)' }}>{prim[k]}</p>
            <p style={{ margin: 0, fontSize: 'var(--type-14-size)', color: 'var(--text-secondary)' }}>{role}</p>
          </div>
        </div>
      ))}
    </div>

    <h2 style={h2Style}>Bearings</h2>
    <p style={pStyle}>The parts of life an item belongs to. Each colour has its own fill and edge token for its pill, and always travels with its word.</p>
    <table style={{ borderCollapse: 'collapse', maxWidth: 720, width: '100%' }}>
      <thead><tr><th style={headCell}>Bearing</th><th style={headCell}>Colour</th><th style={headCell}>Fill</th><th style={headCell}>Edge</th></tr></thead>
      <tbody>
        {bearings.map((b) => (
          <tr key={b}>
            <td style={{ ...cell, color: prim[`bearing/${b}`], fontWeight: 600 }}>{b}</td>
            <td style={cell}><span style={{ display: 'flex', gap: 8, alignItems: 'center' }}><Swatch hex={prim[`bearing/${b}`]} size={24} /><code>{prim[`bearing/${b}`]}</code></span></td>
            <td style={cell}><span style={{ display: 'flex', gap: 8, alignItems: 'center' }}><Swatch hex={dark[`bearing/${b}/fill`]} size={24} /><code>{dark[`bearing/${b}/fill`]}</code></span></td>
            <td style={cell}><span style={{ display: 'flex', gap: 8, alignItems: 'center' }}><Swatch hex={dark[`bearing/${b}/edge`]} size={24} /><code>{dark[`bearing/${b}/edge`]}</code></span></td>
          </tr>
        ))}
      </tbody>
    </table>

    <h2 style={h2Style}>Semantic tokens</h2>
    <p style={pStyle}>Components read these, never the primitives. Alpha tokens are shown over navy/950, the way they are used.</p>
    <table style={{ borderCollapse: 'collapse', maxWidth: 720, width: '100%' }}>
      <thead><tr><th style={headCell}>Token</th><th style={headCell}>Value</th></tr></thead>
      <tbody>
        {semantic.map((k) => (
          <tr key={k}>
            <td style={{ ...cell, fontFamily: 'var(--font-mono)' }}>{k}</td>
            <td style={cell}><span style={{ display: 'flex', gap: 8, alignItems: 'center' }}><Swatch hex={resolve(dark[k])} size={24} /><code>{dark[k]}</code></span></td>
          </tr>
        ))}
      </tbody>
    </table>

    <h2 style={h2Style}>Rules</h2>
    <ul style={{ ...pStyle, paddingLeft: 20 }}>
      <li>Gold is rare: one gold action per view, plus the logo. A state edge (a selected choice) is full gold; decorative edges use the nudge edge.</li>
      <li>Cream means paper: a cream surface only holds something that came from the book.</li>
      <li>Never colour alone: bearings carry their word, marks have their own shapes.</li>
      <li>Every pair is measured over the worst case, glass over the brightest point of the gold glow. See Accessibility for the numbers.</li>
    </ul>
    <p style={{ ...pStyle, fontSize: 'var(--type-14-size)' }}>
      {`Dormant: ${dormant.length} tag-text tokens from version 1 and the light theme values stay in tokens.json for a later light pass; no v2 component reads them.`}
    </p>
  </div>
</Unstyled>
